local function ShowLanguageChangePopup(callback)
    LangPopupOverlay.Visible = true
    LangPopupOverlay.BackgroundTransparency = 1
    LangPopupBox.Size = UDim2.new(0, 250, 0, 140)
    LangPopupBox.BackgroundTransparency = 1
    LoadingBarFill.Size = UDim2.new(0, 0, 1, 0)
    LangSubText.Text = "Language testing in progress."

    TweenService:Create(LangPopupOverlay, TweenInfo.new(0.2, Enum.EasingStyle.Quad), {BackgroundTransparency = 0.5}):Play()
    TweenService:Create(LangPopupBox, TweenInfo.new(0.3, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
        Size = UDim2.new(0, 320, 0, 180),
        BackgroundTransparency = 0.05
    }):Play()

    local loadingDuration = 2.2
    TweenService:Create(LoadingBarFill, TweenInfo.new(loadingDuration, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Size = UDim2.new(1, 0, 1, 0)
    }):Play()

    task.delay(0.9, function()
        if LangPopupOverlay.Visible then
            LangSubText.Text = "Suitable language"
        end
    end)

    task.delay(1.7, function()
        if LangPopupOverlay.Visible then
            LangSubText.Text = "Success"
        end
    end)

    task.delay(loadingDuration + 0.2, function()
        TweenService:Create(LangPopupOverlay, TweenInfo.new(0.2, Enum.EasingStyle.Quad), {BackgroundTransparency = 1}):Play()
        local tw = TweenService:Create(LangPopupBox, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
            Size = UDim2.new(0, 250, 0, 140),
            BackgroundTransparency = 1
        })
        tw:Play()
        tw.Completed:Wait()
        LangPopupOverlay.Visible = false
        if callback then callback() end
    end)
end

local fixLagActive = false
FixLagBtn.MouseButton1Click:Connect(function()
    fixLagActive = not fixLagActive
    if fixLagActive then
        TweenService:Create(SwitchBg, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(255, 255, 255)}):Play()
        TweenService:Create(SwitchKnob, TweenInfo.new(0.2), {
            Position = UDim2.new(1, -18, 0.5, -8),
            BackgroundColor3 = Color3.fromRGB(10, 10, 10)
        }):Play()
        pcall(function()
            settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
            Lighting.GlobalShadows = false
            Lighting.Brightness = 2
            for _, v in pairs(Workspace:GetDescendants()) do
                if v:IsA("BasePart") then
                    v.Material = Enum.Material.SmoothPlastic
                    v.Reflectance = 0
                elseif v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Fire") or v:IsA("Smoke") then
                    v.Enabled = false
                end
            end
        end)
        ShowNotification("Fix Lag", _txt("NotifFixOn"), true)
    else
        TweenService:Create(SwitchBg, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(60, 60, 60)}):Play()
        TweenService:Create(SwitchKnob, TweenInfo.new(0.2), {
            Position = UDim2.new(0, 2, 0.5, -8),
            BackgroundColor3 = Color3.fromRGB(200, 200, 200)
        }):Play()
        pcall(function()
            Lighting.GlobalShadows = true
            Lighting.Brightness = 1
        end)
        ShowNotification("Fix Lag", _txt("NotifFixOff"), false)
    end
end)

local lastUpdate = 0
RunService.RenderStepped:Connect(function(dt)
    if tick() - lastUpdate >= 0.5 then
        lastUpdate = tick()
        local fps = math.floor(1 / dt)
        local ping = 0
        pcall(function()
            ping = math.floor(LocalPlayer:GetNetworkPing() * 1000)
        end)
        StatsText.Text = string.format("FPS: %d | Ping: %dms", fps, ping)
    end
end)

    if (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) and isDraggingOpenBtn then

        if delta.Magnitude > 5 then hasMoved = true end

    end
end)

        OpenButton.Visible = true

            Size = UDim2.new(0, 44, 0, 44)
        }):Play()
    end
end

local function CreateTab(tabKey)
    local tabBtn = Instance.new("TextButton")
    tabBtn.Name = tabKey .. "Btn"
    tabBtn.Parent = TabBar
    tabBtn.BackgroundTransparency = 1
    tabBtn.BorderSizePixel = 0
    tabBtn.Size = UDim2.new(1, 0, 0, 36)
    tabBtn.Font = Enum.Font.GothamBold
    tabBtn.Text = "   " .. _txt(tabKey)
    tabBtn.TextColor3 = Color3.fromRGB(150, 150, 150)
    tabBtn.TextSize = 12
    tabBtn.TextXAlignment = Enum.TextXAlignment.Left

    table.insert(languageListeners, function()
        tabBtn.Text = "   " .. _txt(tabKey)
    end)

    local btnCorner = Instance.new("UICorner")
    btnCorner.CornerRadius = UDim.new(0, 8)
    btnCorner.Parent = tabBtn

    local indicator = Instance.new("Frame")
    indicator.Name = "Indicator"
    indicator.Parent = tabBtn
    indicator.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    indicator.BorderSizePixel = 0
    indicator.Position = UDim2.new(0, 2, 0.5, -10)
    indicator.Size = UDim2.new(0, 3, 0, 20)
    indicator.BackgroundTransparency = 1

    local indCorner = Instance.new("UICorner")
    indCorner.CornerRadius = UDim.new(1, 0)
    indCorner.Parent = indicator

    local tabGradient = Instance.new("UIGradient")
    tabGradient.Parent = tabBtn
    tabGradient.Color = ColorSequence.new({
        ColorSequenceKeypoint.new(0.0, Color3.fromRGB(255, 255, 255)),
        ColorSequenceKeypoint.new(0.4, Color3.fromRGB(255, 80, 80)),
        ColorSequenceKeypoint.new(0.6, Color3.fromRGB(255, 80, 80)),
        ColorSequenceKeypoint.new(1.0, Color3.fromRGB(255, 255, 255))
    })
    tabGradient.Offset = Vector2.new(-1.2, 0)

    task.spawn(function()
        while tabBtn and tabBtn.Parent do
            local tweenInfo = TweenInfo.new(2.2, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut)
            local tween1 = TweenService:Create(tabGradient, tweenInfo, {Offset = Vector2.new(1.2, 0)})
            tween1:Play()
            tween1.Completed:Wait()
            
            tabGradient.Offset = Vector2.new(-1.2, 0)
            task.wait(0.8)
        end
    end)

    local tabContent = Instance.new("ScrollingFrame")
    tabContent.Name = tabKey .. "Content"
    tabContent.Parent = ContentArea
    tabContent.BackgroundTransparency = 1
    tabContent.Size = UDim2.new(1, 0, 1, 0)
    tabContent.CanvasSize = UDim2.new(0, 0, 0, 0)
    tabContent.ScrollBarThickness = 2
    tabContent.ScrollBarImageColor3 = Color3.fromRGB(200, 200, 200)
    tabContent.Visible = false

    local listLayout = Instance.new("UIListLayout")
    listLayout.Parent = tabContent
    listLayout.SortOrder = Enum.SortOrder.LayoutOrder
    listLayout.Padding = UDim.new(0, 6)

    listLayout:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(function()
        tabContent.CanvasSize = UDim2.new(0, 0, 0, listLayout.AbsoluteContentSize.Y)
    end)

    tabBtn.MouseButton1Click:Connect(function()
        CloseDropdown()
        for _, t in pairs(tabs) do
            t.Content.Visible = false
            TweenService:Create(t.Button, TweenInfo.new(0.25, Enum.EasingStyle.Quad), {
                TextColor3 = Color3.fromRGB(150, 150, 150),
                BackgroundTransparency = 1
            }):Play()
            TweenService:Create(t.Indicator, TweenInfo.new(0.2), {
                BackgroundTransparency = 1
            }):Play()
        end
        tabContent.Visible = true
        TweenService:Create(tabBtn, TweenInfo.new(0.25, Enum.EasingStyle.Quad), {
            TextColor3 = Color3.fromRGB(255, 255, 255),
            BackgroundTransparency = 0.85
        }):Play()
        TweenService:Create(indicator, TweenInfo.new(0.2), {
            BackgroundTransparency = 0
        }):Play()
    end)

    if not currentTab then
        currentTab = tabKey
        tabContent.Visible = true
        tabBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
        tabBtn.BackgroundTransparency = 0.85
        indicator.BackgroundTransparency = 0
    end

    table.insert(tabs, {Key = tabKey, Button = tabBtn, Content = tabContent, Indicator = indicator})
    return tabContent
end

    switchKnob.Name = "SwitchKnob"
    switchKnob.Parent = switchBg
    switchKnob.BackgroundColor3 = Color3.fromRGB(200, 200, 200)
    switchKnob.BorderSizePixel = 0
    switchKnob.Position = UDim2.new(0, 2, 0.5, -8)
    switchKnob.Size = UDim2.new(0, 16, 0, 16)

    local knobCorner = Instance.new("UICorner")
    knobCorner.CornerRadius = UDim.new(1, 0)
    knobCorner.Parent = switchKnob

    local toggledState = false

    local function TriggerToggle()
        toggledState = not toggledState
        if toggledState then
            TweenService:Create(switchBg, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(255, 255, 255)}):Play()
            TweenService:Create(switchKnob, TweenInfo.new(0.2), {

                Position = UDim2.new(0, 2, 0.5, -8),
                BackgroundColor3 = Color3.fromRGB(200, 200, 200)
            }):Play()
            TweenService:Create(toggleFrame, TweenInfo.new(0.2), {BackgroundTransparency = 0.8}):Play()
            ShowNotification(_txt(textKey), _txt("Off"), false)
        end
        
        if callback then callback(toggledState) end
    end

    toggleFrame.MouseButton1Click:Connect(function()
        CloseDropdown()
        TriggerToggle()
    end)
end

local function CreateDropdownInTab(tabContent, titleKey, optionKeysList, callback)
    local normalHeight = 38
    local optionHeight = 28
    local optionPadding = 4
    local expandedHeight = 38 + (#optionKeysList * (optionHeight + optionPadding)) + 8

    local dropFrame = Instance.new("Frame")
    dropFrame.Name = titleKey
    dropFrame.Parent = tabContent
    dropFrame.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
    dropFrame.BackgroundTransparency = 0.8
    dropFrame.BorderSizePixel = 0
    dropFrame.Size = UDim2.new(1, -4, 0, normalHeight)
    dropFrame.ClipsDescendants = true

    local frameCorner = Instance.new("UICorner")
    frameCorner.CornerRadius = UDim.new(0, 8)
    frameCorner.Parent = dropFrame

    local frameStroke = Instance.new("UIStroke")
    frameStroke.Parent = dropFrame
    frameStroke.Color = Color3.fromRGB(255, 255, 255)
    frameStroke.Transparency = 0.8
    frameStroke.Thickness = 1

    local titleLabel = Instance.new("TextLabel")
    titleLabel.Parent = dropFrame
    titleLabel.BackgroundTransparency = 1
    titleLabel.Position = UDim2.new(0, 12, 0, 0)
    titleLabel.Size = UDim2.new(0.4, 0, 0, 38)
    titleLabel.Font = Enum.Font.GothamMedium
    titleLabel.Text = _txt(titleKey)
    titleLabel.TextColor3 = Color3.fromRGB(240, 240, 240)
    titleLabel.TextSize = 12
    titleLabel.TextXAlignment = Enum.TextXAlignment.Left

    local selectedIndex = 1
    local selectedLabel = Instance.new("TextLabel")
    selectedLabel.Parent = dropFrame
    selectedLabel.BackgroundTransparency = 1
    selectedLabel.Position = UDim2.new(0.4, 0, 0, 0)
    selectedLabel.Size = UDim2.new(0.6, -16, 0, 38)
    selectedLabel.Font = Enum.Font.GothamMedium
    selectedLabel.Text = _txt(optionKeysList[selectedIndex])
    selectedLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
    selectedLabel.TextSize = 11
    selectedLabel.TextXAlignment = Enum.TextXAlignment.Right

    local dropBtn = Instance.new("TextButton")
    dropBtn.Parent = dropFrame
    dropBtn.BackgroundTransparency = 1
    dropBtn.Size = UDim2.new(1, 0, 0, 38)
    dropBtn.Text = ""

    local listHolder = Instance.new("Frame")
    listHolder.Parent = dropFrame
    listHolder.BackgroundTransparency = 1
    listHolder.Position = UDim2.new(0, 6, 0, 38)
    listHolder.Size = UDim2.new(1, -12, 0, #optionKeysList * (optionHeight + optionPadding))

    local listLayout = Instance.new("UIListLayout")
    listLayout.Parent = listHolder
    listLayout.SortOrder = Enum.SortOrder.LayoutOrder
    listLayout.Padding = UDim.new(0, optionPadding)

    local isOpenDrop = false
    local activeIndicator = nil

    for index, optKey in ipairs(optionKeysList) do
        local optBtn = Instance.new("TextButton")
        optBtn.Parent = listHolder
        optBtn.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
        optBtn.BackgroundTransparency = 0.3
        optBtn.BorderSizePixel = 0
        optBtn.Size = UDim2.new(1, 0, 0, optionHeight)
        optBtn.Font = Enum.Font.GothamMedium
        optBtn.Text = "   " .. _txt(optKey)
        optBtn.TextColor3 = Color3.fromRGB(220, 220, 220)
        optBtn.TextSize = 11
        optBtn.TextXAlignment = Enum.TextXAlignment.Left

        local optCorner = Instance.new("UICorner")
        optCorner.CornerRadius = UDim.new(0, 6)
        optCorner.Parent = optBtn

        local indicatorLine = Instance.new("Frame")
        indicatorLine.Name = "IndicatorLine"
        indicatorLine.Parent = optBtn
        indicatorLine.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
        indicatorLine.BorderSizePixel = 0
        indicatorLine.Position = UDim2.new(0, 6, 1, -3)
        indicatorLine.Size = UDim2.new(1, -12, 0, 2)
        indicatorLine.BackgroundTransparency = 1

        local indLineCorner = Instance.new("UICorner")
        indLineCorner.CornerRadius = UDim.new(1, 0)
        indLineCorner.Parent = indicatorLine

        if index == 1 then
            indicatorLine.BackgroundTransparency = 0
            activeIndicator = indicatorLine
        end

        table.insert(languageListeners, function()
            titleLabel.Text = _txt(titleKey)
            optBtn.Text = "   " .. _txt(optKey)
            selectedLabel.Text = _txt(optionKeysList[selectedIndex])
        end)

        optBtn.MouseButton1Click:Connect(function()
            selectedIndex = index
            selectedLabel.Text = _txt(optKey)
            isOpenDrop = false
            TweenService:Create(dropFrame, TweenInfo.new(0.25, Enum.EasingStyle.Quad), {Size = UDim2.new(1, -4, 0, normalHeight)}):Play()
            
            if activeIndicator then
                TweenService:Create(activeIndicator, TweenInfo.new(0.2), {BackgroundTransparency = 1}):Play()
            end
            indicatorLine.BackgroundTransparency = 0
            activeIndicator = indicatorLine

            if titleKey == "SelectLang" then
                ShowLanguageChangePopup(function()
                    currentLang = optKey
                    for _, listener in ipairs(languageListeners) do
                        pcall(listener)
                    end
                    if optKey == "Tiếng Việt" then
                        ShowNotification("Language", _txt("LangNotifVN"), true)
                    else
                        ShowNotification("Language", _txt("LangNotifEN"), true)
                    end
                end)
            else
                ShowNotification(_txt("System"), _txt("SysSelected") .. _txt(optKey), true)
                if callback then callback(optKey) end
            end
        end)
    end

    dropBtn.MouseButton1Click:Connect(function()
        CloseDropdown()
        isOpenDrop = not isOpenDrop
        local targetSize = isOpenDrop and expandedHeight or normalHeight
        TweenService:Create(dropFrame, TweenInfo.new(0.25, Enum.EasingStyle.Quad), {Size = UDim2.new(1, -4, 0, targetSize)}):Play()
    end)
end
